MODULE: Dockerfile
PATH: d:\newfeed\services\auth-service\Dockerfile

FROM golang:1.21-alpine AS builder

WORKDIR /build

# Copy go mod files
COPY go.mod go.sum ./

# Download dependencies
RUN go mod download

# Copy source code
COPY . .

# Build the application
RUN CGO_ENABLED=0 GOOS=linux go build -o auth-service main.go

# Final stage
FROM alpine:latest

RUN apk --no-cache add ca-certificates

WORKDIR /root/

# Copy the binary from builder
COPY --from=builder /build/auth-service .

# Expose ports
EXPOSE 50051 8080

# Run the service
CMD ["./auth-service"]
