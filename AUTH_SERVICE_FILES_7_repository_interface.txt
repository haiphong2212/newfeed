package repository

import "github.com/newfeed/services/auth-service/internal/domain"

// UserRepository defines the interface for user data access
type UserRepository interface {
	// Create saves a new user to the repository
	Create(user *domain.User) error
	// GetByID retrieves a user by ID
	GetByID(id string) (*domain.User, error)
	// GetByEmail retrieves a user by email
	GetByEmail(email string) (*domain.User, error)
	// Update updates an existing user
	Update(user *domain.User) error
	// Delete removes a user
	Delete(id string) error
	// List retrieves all users with pagination
	List(limit, offset int) ([]*domain.User, error)
}

// TokenRepository defines the interface for token data access
type TokenRepository interface {
	// Create saves a new token to the repository
	Create(token *domain.Token) error
	// GetByToken retrieves a token by its string value
	GetByToken(tokenString string) (*domain.Token, error)
	// GetByUserID retrieves tokens for a user
	GetByUserID(userID string) ([]*domain.Token, error)
	// Delete removes a token
	Delete(id string) error
	// DeleteExpired removes expired tokens
	DeleteExpired() error
	// ValidateToken checks if a token exists and is not expired
	ValidateToken(tokenString string) (bool, error)
}
